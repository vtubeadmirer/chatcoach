```powershell
cd C:\Users\dlem9\dev\chatcoach

$P
New-Item -ItemType Directory -Force C:\Users\dlem9\dev\chatcoach-data\audio, C:\Users\dlem9\dev\chatcoach-data\reports | Out-Null
```
